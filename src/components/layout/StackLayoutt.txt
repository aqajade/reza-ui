// src/components/layout/StackLayout.tsx
import * as React from 'react';

import { cn } from '@/lib/utils';

type StackLayoutProps = React.ComponentProps<'div'> & {
  /** Arah barisan stack: 'col' atau 'row' */
  direction?: 'col' | 'row';
  /** 🟢 TERPUSAT MAKRO: Menggantikan stackClass lama untuk pembungkus terluar layout */
  mainStackClass?: string;
  /** 🟢 TERPUSAT MIKRO: Untuk memodularkan wadah kolom teks/konten internal (e.g., bg-primary-100) */
  subStackClass?: string;
};

export function StackLayout({
  className,
  direction = 'col',
  mainStackClass,
  subStackClass,
  children,
  ...props
}: StackLayoutProps) {
  return (
    <div
      className={cn(
        'flex',
        direction === 'col' ? 'flex-col' : 'flex-row',
        'stack-layout', // Mengunci kelas dasar globals.css
        mainStackClass, // Suntikan gap makro clamp dari data
        className
      )}
      {...props}
    >
      {/* 
        Jika subStackClass diisi dari data, kita otomatis membungkus 
        children di dalam wadah kontainer mikro yang modular!
      */}
      {subStackClass ? (
        <div className={cn('flex w-full flex-col', subStackClass)}>
          {children}
        </div>
      ) : (
        children
      )}
    </div>
  );
}
