// src/components/layout/card/CardHeader.tsx
import * as React from 'react';

import { cn } from '@/lib/utils';

type CardHeaderProps = React.ComponentProps<'div'> & {
  cardTitle: string;
  cardSubtitle?: string;
  cardDescriptionIntro?: string;
  cardDescriptionOutro?: string;
  cardHeaderClass?: string;
  headingClass?: string;
  subTitleClass?: string;
  descriptionIntroClass?: string;
  descriptionOutroClass?: string;
  titleElement?: 'h2' | 'h3' | 'h4'| 'h5';
  subTitleElement?: 'h2' | 'h3' | 'h4' | 'h5';
  paragraphElement?: 'p' | 'span' | 'div';
};

export function CardHeader({
  className,
  cardHeaderClass,
  cardTitle,
  cardSubtitle,
  cardDescriptionIntro,
  cardDescriptionOutro,
  headingClass,
  subTitleClass,
  descriptionIntroClass,
  descriptionOutroClass,
  titleElement = 'h3',
  subTitleElement = 'h4',
  paragraphElement = 'p',
  ...props
}: CardHeaderProps) {
  const HeadingComponent = titleElement;
  const SubTitleComponent = subTitleElement;
  const ParagraphComponent = paragraphElement;

  // Helper reuseable untuk memecah teks dan menyelipkan span font-bold kustom
  const parseHighlightText = (text: string, defaultClass: string) => {
    if (!text.includes('[') || !text.includes(']')) {
      return (
        <ParagraphComponent className={defaultClass}>{text}</ParagraphComponent>
      );
    }

    return (
      <ParagraphComponent className={defaultClass}>
        {text.split(/(\[.*?\])/).map((part, index) => {
          if (part.startsWith('[') && part.endsWith(']')) {
            const cleanWord = part.slice(1, -1);
            return (
              // 🛠️ SUNTIKKAN SPAN TEBAL DI SINI: Otomatis aktif saat mendeteksi kurung siku!
              <span key={index} className='font-extrabold'>
                {cleanWord}
              </span>
            );
          }
          return part;
        })}
      </ParagraphComponent>
    );
  };

  return (
    <div className={cn('flex flex-col', cardHeaderClass, className)} {...props}>
      {/* 1. CARD TITLE */}
      <HeadingComponent
        className={cn(
          'font-bold tracking-tight text-neutral-900',
          headingClass
        )}
      >
        {cardTitle}
      </HeadingComponent>

      {/* 2. CARD SUBTITLE */}
      {cardSubtitle && (
        <SubTitleComponent
          className={cn('font-medium text-neutral-700', subTitleClass)}
        >
          {cardSubtitle}
        </SubTitleComponent>
      )}

      {/* 3. CARD DESCRIPTION INTRO (Sudah mendukung auto-highlight span) */}
      {cardDescriptionIntro &&
        parseHighlightText(
          cardDescriptionIntro,
          cn('font-medium text-neutral-600', descriptionIntroClass)
        )}

      {/* 4. CARD DESCRIPTION END (Sudah mendukung auto-highlight span) */}
      {cardDescriptionOutro &&
        parseHighlightText(
          cardDescriptionOutro,
          cn('font-normal text-neutral-500', descriptionOutroClass)
        )}
    </div>
  );
}
