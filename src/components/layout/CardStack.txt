// src/components/layout/card/CardStack.tsx
import * as React from 'react';

import { cn } from '@/lib/utils';

type CardStackProps = React.ComponentProps<'div'> & {
  /** Menampung string gabungan rumus TEXTJOIN Excel kamu (e.g., 'gap-5 items-center text-center') */
  cardStackClass?: string;
};

export function CardStack({
  className,
  cardStackClass,
  ...props
}: CardStackProps) {
  return (
    <div
      // 'card-stack' adalah baju dasar, 'cardStackClass' adalah suntikan dinamis rumus Excel kamu
      className={cn('card-stack', cardStackClass, className)}
      {...props}
    />
  );
}
