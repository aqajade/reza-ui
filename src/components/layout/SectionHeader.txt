// src/components/layout/SectionHeader.tsx
import * as React from 'react';

import { cn } from '@/lib/utils';

/**
 * 1. KONTRAK TYPE: DI-UPGRADE UNTUK MENDUKUNG KELAS TEKS DIVERSE 🟢
 */
type SectionHeaderProps = React.ComponentProps<'div'> & {
  /** Isi teks untuk judul utama section */
  title: string;

  /** Isi teks untuk deskripsi/paragraf bawah (Opsional) */
  description?: string;

  /** SEMANTIK: Mengubah wujud tag judul murni lewat nama properti 'headingElement' */
  headingElement?: 'h1' | 'h2' | 'h3' | 'h4';

  /** 🛠️ BARU: Suntikan kelas kustom untuk heading dari data Excel / page-data.ts */
  headingClass?: string;

  /** 🛠️ BARU: Suntikan kelas kustom untuk paragraf dari data Excel / page-data.ts */
  paragraphClass?: string;

  /** Menyediakan ID kustom untuk accessibility aria-labelledby form */
  headingId?: string;

  /** Menyediakan ID kustom untuk accessibility aria-describedby form */
  descriptionId?: string;
};

/**
 * 2. KOMPONEN UTAMA JAUH LEBIH BERSIH & BERKASA (PASCALCASE)
 */
export function SectionHeader({
  className,
  title,
  description,
  headingElement = 'h2', // Langsung ditangkap & diberi nilai default 'h2' dengan suci!
  headingClass, // 🛠️ Tangkap properti baru kamu di sini
  paragraphClass, // 🛠️ Tangkap properti baru kamu di sini
  headingId,
  descriptionId,
  ...props
}: SectionHeaderProps) {
  // React mewajibkan huruf pertama variabel tag dinamis berupa Huruf Kapital (PascalCase)
  const HeadingComponent = headingElement;

  return (
    // 'section-header' adalah kelas dasar di globals.css yang murni mengurus arah flex-col
    <div className={cn('section-header', className)} {...props}>
      {/* 🟢 JUDUL UTAMA: Menggabungkan kelas default dan paksaan kelas kustom dari Hero via cn() */}
      <HeadingComponent
        id={headingId}
        className={cn('section-heading', headingClass)}
      >
        {title}
      </HeadingComponent>

      {/* 🟢 PARAGRAF DESKRIPSI: Menggabungkan kelas default dan paksaan kelas kustom dari Hero via cn() */}
      {description && (
        <p
          id={descriptionId}
          className={cn('section-paragraph', paragraphClass)}
        >
          {description}
        </p>
      )}
    </div>
  );
}
