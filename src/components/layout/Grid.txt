import * as React from 'react';

import { cn } from '@/lib/utils'; // Panggil fungsi penggabung sakti hasil init Shadcn UI

type GridProps = React.ComponentProps<'div'> & {
  /** Suntikan rumus gridClass dinamis dari Excel / page-data.ts */
  gridClass?: string;
};

export function Grid({ className, gridClass, ...props }: GridProps) {
  return (
    <div
      className={cn(
        'grid', // Baju dasar wajib mengaktifkan ekosistem CSS Grid layout
        gridClass, // Pasokan string kolom bawaan data pusat Excel
        className // Gerbang dadakan dari file komponen luar (.tsx) selalu menang di paling belakang!
      )}
      {...props}
    />
  );
}
