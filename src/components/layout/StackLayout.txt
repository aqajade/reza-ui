// src/components/layout/StackLayout.tsx
import * as React from 'react';

import { cn } from '@/lib/utils';

/**
 * 1. DEFINISI TIPE PROPS (BIP MODERN USING TYPE)
 * Menggabungkan properti bawaan dari tag HTML <div> asli
 * dengan properti kustom untuk flex configuration opsional.
 */
type StackLayoutProps = React.ComponentProps<'div'> & {
  /** Properti opsional jika ingin mengubah arah stack (default: 'col') */
  direction?: 'col' | 'row';
  /** 🟢 BARU & BIP: Menyerap langsung string kelas jarak gap kustom clamp dari Excel */
  stackClass?: string;
};

/**
 * 2. KOMPONEN UTAMA STACKLAYOUT (PASCALCASE)
 */
export function StackLayout({
  className,
  stackClass,
  direction = 'col', // Defaultnya vertical stack sesuai kebutuhan layout figma kamu
  ...props
}: StackLayoutProps) {
  return (
    <div
      className={cn(
        // Hukum dasar: Wajib Flexbox dan arah vertikal/horizontal
        'flex',
        direction === 'col' ? 'flex-col' : 'flex-row',

        // 'stack-layout' adalah kelas dasar SSOT dari globals.css kamu
        // yang mengunci gap-8 (mobile) dan gap-16/20 (desktop) hasil kalkulasi spreadsheet
        'stack-layout',
        stackClass,

        className // Mengizinkan override/penambahan kelas eksternal via cn()
      )}
      {...props}
    />
  );
}
