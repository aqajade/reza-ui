/**
 * ==========================================================================
 * 1. LAYOUT PRIMITIF CONFIG (Mendukung PageSection & CustomContainer)
 * ==========================================================================
 */
export type SectionLayoutConfig = {
  /** ID unik untuk navigasi anchor HTML (e.g., 'header', 'hero', 'services') */
  id: string;
  title: string;
  desc: string;
  /** Class tambahan opsional untuk sekat luar (e.g., 'bg-slate-50', 'sticky top-0') */

  /** Baju container: 'default' (globals.css) atau 'custom' (suntikan rumus px Excel) */

  /** String rumus max-width pecahan ganjil milik Excel kamu */
  containerClass?: string;
  /** Memaksa arah barisan StackLayout: 'col' (vertikal) atau 'row' (horizontal menyamping) */
  layoutStackDirection?: 'col' | 'row';
    contentStackDirection?: 'col' | 'row';
  itemStackDirection?: 'col' | 'row';
  /** Class tambahan opsional untuk sekat luar (e.g., 'bg-slate-50', 'sticky top-0') */

  contentStackClass?: string;
  itemStackClass?: string;
  /** Class u/ section heading */
  headingClass?: string;
  /** Menentukan kedaulatan wujud tag semantik judul luar di HTML: 'h1' | 'h2' | 'h3' */
  headingElement?: 'h1' | 'h2' | 'h3' | 'h4';
  imageContainerClass?: string;
  imageSrc: string | StaticImageData;
  imageClass?: string;
  imageAlt?: string;
};
